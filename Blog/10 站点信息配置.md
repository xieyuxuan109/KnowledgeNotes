```
type QQ struct {
	AppID    string `json:"appID" yaml:"appID"`
	AppKey   string `json:"appKey" yaml:"appKey"`
	Redirect string `json:"redirect" yaml:"redirect"`
}

func (q QQ) Url() string {
	return fmt.Sprintf("https://graph.qq.com/oauth2.0/authorize?client_id=%s&display=pc&redirect_uri=%s&response_type=code&state=random_state_abc&which=Login", q.AppID, q.Redirect)
}
```

## 敏感字段打码返回
```
func (SiteApi) SiteInfoView(c *gin.Context) {
	var cr SiteInfoRequest
	err := c.ShouldBindUri(&cr)
	if err != nil {
		res.FailWithError(err, c)
		return
	}
	var data any
	switch cr.Name {
	case "site":
		data = global.Config.Site
	case "email":
		req := global.Config.Email
		req.AuthCode = "******"
		data = req
	case "qq":
		req := global.Config.QQ
		req.AppKey = "******"
		data = req
	case "qiNiu":
		req := global.Config.QiNiu
		req.SecretKey = "******"
		data = req
	case "ai":
		req := global.Config.Ai
		req.SecretKey = "******"
		data = req
	default:
		res.FailWithMsg("不存在的配置", c)
	}
	res.OkWithData(data, c)
}
```
