## 上传图片
```
package image_api

import (
	"blog_server/common/res"
	"blog_server/utils"
	"errors"
	"fmt"
	"strings"

	"github.com/gin-gonic/gin"
)

var whiteList = []string{
	"jpg",
	"jepg",
	"png",
	"webp",
	"gif",
}

func (ImageApi) ImageUploadsView(c *gin.Context) {
	fileHeader, err := c.FormFile("file")
	if err != nil {
		res.FailWithError(err, c)
		return
	}
	//判断大小
	if fileHeader.Size > 2*1024*1024 {
		res.FailWithMsg("文件大于2MB", c)
	}
	//判断文件名
	err = ImageSuffixJudge(fileHeader.Filename)
	if err != nil {
		res.FailWithError(err, c)
	}
	filePath := fmt.Sprintf("uploads/images/%s", fileHeader.Filename)
	err = c.SaveUploadedFile(fileHeader, filePath)
	if err != nil {
		res.FailWithError(err, c)
		return
	}
	res.Ok("/"+filePath, "图片上传成功", c)
}
func ImageSuffixJudge(fileName string) error {
	_list := strings.Split(fileName, ".")
	if len(_list) == 1 {
		return errors.New("错误的文件名")
	}
	suffix := _list[len(_list)-1]
	if !utils.InList(suffix, whiteList) {
		return errors.New("文件非法")
	}
	return nil
}
```
