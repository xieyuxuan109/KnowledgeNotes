func (SiteApi) SiteUpdateView(c *gin.Context) {
	var cr SiteInfoRequest
	if err := c.ShouldBindUri(&cr); err != nil {
		res.FailWithError(err, c)
		return
	}

	switch cr.Name {
	case "site":
		var req conf.Site
		if err := c.ShouldBindJSON(&req); err != nil {
			res.FailWithError(err, c)
			return
		}
		UpdateSite(req)

	case "email":
		var req conf.Email
		if err := c.ShouldBindJSON(&req); err != nil {
			res.FailWithError(err, c)
			return
		}
		// 保留原有的密码（前端用 ****** 占位）
		if req.AuthCode == "******" {
			req.AuthCode = global.Config.Email.AuthCode
		}
		global.Config.Email = req

	case "qq":
		var req conf.QQ
		if err := c.ShouldBindJSON(&req); err != nil {
			res.FailWithError(err, c)
			return
		}
		if req.AppKey == "******" {
			req.AppKey = global.Config.QQ.AppKey
		}
		global.Config.QQ = req

	case "qiNiu":
		var req conf.QiNiu
		if err := c.ShouldBindJSON(&req); err != nil {
			res.FailWithError(err, c)
			return
		}
		if req.AccessKey == "******" {
			req.AccessKey = global.Config.QiNiu.AccessKey
		}
		global.Config.QiNiu = req

	case "ai":
		var req conf.Ai
		if err := c.ShouldBindJSON(&req); err != nil {
			res.FailWithError(err, c)
			return
		}
		if req.SecretKey == "******" {
			req.SecretKey = global.Config.Ai.SecretKey
		}
		global.Config.Ai = req

	default:
		res.FailWithMsg("不存在的配置", c)
		return
	}
	core.SetConf()
	res.OkWithMsg("更新成功", c)
}
```
## 持久化更新配置
```
func SetConf() {
	byteData, err := yaml.Marshal(global.Config)
	if err != nil {
		logrus.Errorf("conf读取失败 %s", err)
	}
	err = os.WriteFile(flags.FlagOptions.File, byteData, 0666)
	if err != nil {
		logrus.Errorf("conf更新失败 %s", err)
	}
	fmt.Printf("修改配置文件成功 %s 成功！\n", flags.FlagOptions.File)
}
