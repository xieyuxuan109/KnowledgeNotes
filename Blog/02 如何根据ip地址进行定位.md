## demo
```
package main

import (
	"fmt"
	"log"

	"github.com/lionsoul2014/ip2region/binding/golang/xdb"
)

func main() {
	// 数据文件路径
	dbPath := "../init/ip2region_v4.xdb"

	// 创建基于文件的查询对象
	searcher, err := xdb.NewWithFileOnly(xdb.IPv4, dbPath)
	if err != nil {
		log.Fatalf("加载 ip2region 数据库失败: %v", err)
	}
	defer searcher.Close()

	// 查询 IP 地址
	ip := "175.0.201.207"
	region, err := searcher.Search(ip)
	if err != nil {
		log.Fatalf("查询 IP (%s) 失败: %v", ip, err)
	}

	fmt.Printf("IP: %s, 地理位置: %s\n", ip, region)
}
```
