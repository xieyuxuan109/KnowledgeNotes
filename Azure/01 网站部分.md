## 用powershell创建虚拟机
```
$ResourceGroupNameVM=xxxx;
$ResourceGroupNameVnet=xxxx;
$AzVirtualNetworkName=xxx;
$AzVirtualNetworkSubnetName=xxx;
$VMName=xxx;
$securityTypeStnd="Standard"
$NICName="$VMName-NIC"
$VMSize = "Standard_D2ls_v5"
$Location=xxx;
$user="testhh";
$securePassword=ConvertTo-SecureString -String "Testhh123456" -AsPlainText -Force;
$ImageReferenceId='MicrosoftWindowsServer:WindowsServer:2022-datacenter-azure-edition:latest'
$Vnet=Get-AzVirtualNetwork -Name $AzVirtualNetworkName -ResourceGroupName $ResourceGroupNameVnet;
$Subnet=Get-AzVirtualNetworkSubnetConfig -VirtualNetwork $Vnet -Name AzVirtualNetworkSubnetName;

$NIC=New-AzNetworkInterface -Name $NICName -ResourceGroupName $ResourceGroupNameVM -Location $Location; 
$cred = New-Object System.Management.Automation.PSCredential ($user,$securePassword);
$VirtualMachine = New-AzVMConfig -VMName $VMName -VMSize $VMSize -securityType $securityTypeStnd;
$VirtualMachine = Set-AzVMOperatingSystem -VM $VirtualMachine -Windows -ComputerName $VMName -Credential $cred; 
$VirtualMachine=Add-AzVMNetworkInterface -VM $VirtualMachine -Id $NIC.Id
$VirtualMachine=Set-AzVMSourceImage -VM $VirtualMachine -Id $ImageReferenceId
$VirtualMachine = SetAzBootDiagnostic -VM $VirtualMachine -Disable
New-AzVm
    -ResourceGroupName $ResourceGroupNameVM `
    <!-- -Name  $VMName ` -->
    -Location  $Location`
    -VM $VirtualMachine -Verbose
    <!-- -VMSize = $VMSize `
    -ImageReferenceId $ImageReferenceId `
    -VirtualNetworkName $VirtualNetworkName `
    -SubnetId $Subnet.Id
    -OpenPorts 3389 ` -->
    
```

## bicep创建web app plan
```
param resourceName string
param location string

resource serverfarm 'Microsoft.Web/serverfarms@2022-09-01' = {
  name: resourceName
  location: location
  sku: {
    name: 'F1'
  }
  properties: {
    hyperV: false
    perSiteScaling: false
    reserved: false
    zoneRedundant: false
  }
}
output serverfarmId string = serverfarm.id
```

## 创建虚拟机快照
```

恢复 ： 点击创建磁盘 => 选择快照方式添加 => create VM => 不会重新创建账号密码 => 使用之前的账号密码
```
