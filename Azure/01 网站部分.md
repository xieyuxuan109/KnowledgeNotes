## 用powershell创建虚拟机
```
$Location=xxx;
$ResourceGroupName=xxxx;
$user="testhh";
$securePassword=ConvertTo-SecureString -String "Testhh123456" -AsPlainText -Force;
$cred = New-Object System.Management.Automation.PSCredential ($user,$securePassword);
$Vet=xxx;
$NIC=New-AzNetworkInterface -Name $NICName -ResourceGroupName $ResourceGroupName -Location $Location 
New-AzVm `  
    -ResourceGroupName 'xxxxxxx' `
    -Name 'xxxx' `
    -Location 'xxxxx' `
    -VMSize = "Standard_DS3" `
    -ImageReferenceId 'MicrosoftWindowsServer:WindowsServer:2022-datacenter-azure-edition:latest' `
    -VirtualNetworkName 'myVnet' `
    (-SubnetName 'mySubnet' `)
    -SubnetId $Vnet.Subbets[0].Id
    -OpenPorts 80,3389 `
    -Credential $cred
```
